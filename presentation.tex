\section{Génération du Graphe G(n, p) — Modèle Erdős–Rényi}

\subsection{Explication}

\begin{tcolorbox}[explication]
Le modèle \textbf{Erdős–Rényi G(n, p)} génère aléatoirement un graphe non orienté
de $n$ nœuds. Pour \textbf{chaque paire} de nœuds distincts $(i,\, j)$ avec $i < j$,
on tire un réel aléatoire $r \in [0,1]$ selon une loi uniforme :
\[
  (i,j) \in E \iff r \leq p
\]
La contrainte $i < j$ garantit l'absence de boucles ($i \neq j$) et
la non-duplication des arêtes (non orienté).

\medskip
\textbf{Propriétés attendues :}
\begin{itemize}
  \item Nombre moyen d'arêtes : $\displaystyle\mathbb{E}[|E|] = \binom{n}{2}\cdot p = \frac{n(n-1)}{2}\cdot p$
  \item Pour $n=50$, $p=0{,}10$ : environ $\mathbf{122}$ arêtes.
  \item Le graphe peut être non connexe — ce sera traité par l'algorithme 2.
\end{itemize}
\end{tcolorbox}

\subsection{Algorithme}

\begin{algorithm}[H]
\caption{Génération G(n, p) — Erdős–Rényi}
\KwIn{$n \in \mathbb{N}^*$ : nombre de nœuds,\quad $p \in [0,1]$ : probabilité}
\KwOut{$G$ : graphe non orienté, sans boucle}
\BlankLine
$G \leftarrow$ graphe vide avec les sommets $\{0, 1, \ldots, n-1\}$\;
\Pour{$i \leftarrow 0$ \textbf{ à } $n - 2$}{
    \Pour{$j \leftarrow i + 1$ \textbf{ à } $n - 1$}{
        $r \leftarrow \mathcal{U}(0,\;1)$ \com{tirage uniforme dans $[0,1]$}
        \Si{$r \leq p$}{
            Ajouter l'arête $(i,\; j)$ à $G$\;
        }
    }
}
\Retourner $G$\;
\end{algorithm}

\subsection{Code Python}

\begin{lstlisting}[style=python,
  caption={Génération du graphe — méthode \texttt{\_generate()}}]
def _generate(self, n: int) -> None:
    # Génération G(n, p) — modèle Erdos-Renyi
    # NetworkX parcourt toutes les paires (i,j) avec i < j
    # et ajoute chaque arête avec probabilité self.P = 0.10
    G = nx.erdos_renyi_graph(n, self.P)
\end{lstlisting}

\begin{tcolorbox}[attention]
\texttt{nx.erdos\_renyi\_graph(n, p)} implémente exactement l'algorithme ci-dessus.
La bibliothèque garantit l'absence de boucles et la non-orientation du graphe.
\end{tcolorbox}


\section{Garantie de Connexité — BFS et Fusion de Composantes}

\subsection{Explication}

\begin{tcolorbox}[explication]
Un graphe G(n,p) n'est pas toujours connexe. La connexité est garantie pour
$p \geq \frac{\ln n}{n}$, mais pour $p = 0{,}10$ et $n$ petit, des nœuds
peuvent rester isolés.

\medskip
\textbf{Stratégie en deux étapes :}
\begin{enumerate}
  \item \textbf{BFS (Breadth-First Search)} : parcours en largeur depuis
        chaque nœud non visité pour identifier les \emph{composantes connexes}.
  \item \textbf{Fusion} : tant qu'il y a plus d'une composante, on choisit
        un nœud aléatoire dans chacune des deux premières et on les relie
        par une nouvelle arête.
\end{enumerate}

\medskip
Au plus $k-1$ arêtes sont ajoutées pour $k$ composantes initiales.
\end{tcolorbox}

\subsection{Algorithme}

\subsubsection*{Étape A — BFS : trouver les composantes connexes}

\begin{algorithm}[H]
\caption{BFS — Identification des composantes connexes}
\KwIn{$G$ : graphe, $n$ : nombre de nœuds}
\KwOut{$\mathit{composantes}$ : liste de listes de nœuds}
\BlankLine
$\mathit{visité}[v] \leftarrow \Faux$ pour tout $v \in \{0, \ldots, n-1\}$\;
$\mathit{composantes} \leftarrow [\;]$\;
\Pour{$s \leftarrow 0$ \textbf{ à } $n - 1$}{
    \Si{$\mathit{visité}[s] = \Faux$}{
        $c \leftarrow [\;]$ \com{nœuds de la composante courante}
        $\mathit{file} \leftarrow [s]$\;
        $\mathit{visité}[s] \leftarrow \Vrai$\;
        \TantQue{$\mathit{file} \neq \varnothing$}{
            $u \leftarrow$ \textsc{Défiler}($\mathit{file}$)\;
            Ajouter $u$ à $c$\;
            \Pour{chaque voisin $v$ de $u$ dans $G$}{
                \Si{$\mathit{visité}[v] = \Faux$}{
                    $\mathit{visité}[v] \leftarrow \Vrai$\;
                    \textsc{Enfiler}($v$,\; $\mathit{file}$)\;
                }
            }
        }
        Ajouter $c$ à $\mathit{composantes}$\;
    }
}
\Retourner $\mathit{composantes}$\;
\end{algorithm}

\subsubsection*{Étape B — Fusion des composantes}

\begin{algorithm}[H]
\caption{Fusion — Garantie de connexité du graphe}
\KwIn{$G$ : graphe, $\mathit{composantes}$ : liste des composantes}
\KwOut{$G$ : graphe connexe}
\BlankLine
\TantQue{$|\mathit{composantes}| > 1$}{
    $c_1 \leftarrow \mathit{composantes}[0]$ \com{première composante}
    $c_2 \leftarrow \mathit{composantes}[1]$ \com{deuxième composante}
    $u \leftarrow$ \textsc{ChoixAléatoire}$(c_1)$\;
    $v \leftarrow$ \textsc{ChoixAléatoire}$(c_2)$\;
    Ajouter l'arête $(u,\; v)$ à $G$\;
    \com{Recalculer les composantes après ajout}
    $\mathit{composantes} \leftarrow$ \textsc{BFS}$(G)$\;
}
\Retourner $G$\;
\end{algorithm}

\subsection{Code Python}

\begin{lstlisting}[style=python,
  caption={Garantie de connexité — méthode \texttt{\_generate()}}]
# Étape A : BFS via NetworkX pour trouver les composantes
components = list(nx.connected_components(G))

# Étape B : fusion des composantes jusqu'a connexité totale
while len(components) > 1:
    c1 = list(components[0])   # premiere composante
    c2 = list(components[1])   # deuxième composante
    # Relier un noeud aléatoire de c1 a un noeud aléatoire de c2
    G.add_edge(random.choice(c1), random.choice(c2))
    # Recalculer les composantes apres l'ajout de l'arête
    components = list(nx.connected_components(G))
\end{lstlisting}


\section{Coloration Gloutonne du Graphe}

\subsection{Explication}

\begin{tcolorbox}[explication]
La \textbf{coloration de graphe} attribue un entier (couleur) à chaque nœud
de façon que \textbf{deux nœuds adjacents n'aient jamais la même couleur}.
Formellement, une $k$-coloration valide est une fonction :
\[
  f : V \to \{0, 1, \ldots, k-1\} \quad \text{tel que} \quad
  \forall (u,v) \in E,\; f(u) \neq f(v)
\]

\medskip
L'\textbf{algorithme glouton} (\textit{Greedy Coloring}) avec la stratégie
\textit{largest\_first} (nœuds de degré le plus élevé traités en premier) :
\begin{enumerate}
  \item Trie les nœuds par degré décroissant.
  \item Pour chaque nœud $v$, collecte les couleurs déjà utilisées par
        ses voisins déjà colorés.
  \item Attribue la \textbf{plus petite couleur disponible} non utilisée.
\end{enumerate}

\medskip
L'algorithme ne garantit pas le nombre chromatique optimal, mais produit
une coloration valide efficacement en $O(n + |E|)$.
\end{tcolorbox}

\subsection{Algorithme}

\begin{algorithm}[H]
\caption{Coloration gloutonne — \textit{Greedy Coloring}}
\KwIn{$G=(V,E)$ : graphe connexe}
\KwOut{$\mathit{couleur}$ : $V \to \mathbb{N}$, coloration valide}
\BlankLine
Trier $V$ par degré décroissant $\rightarrow$ ordre $[v_0, v_1, \ldots, v_{n-1}]$\;
\Pour{chaque sommet $v \in V$}{
    $\mathit{couleur}[v] \leftarrow -1$ \com{non coloré}
}
\Pour{chaque sommet $v$ dans l'ordre trié}{
    \com{Collecter les couleurs des voisins déjà colorés}
    $\mathit{utilisées} \leftarrow
    \bigl\{\mathit{couleur}[u] \;\big|\; u \in \mathcal{N}(v),\;
    \mathit{couleur}[u] \geq 0\bigr\}$\;
    $c \leftarrow 0$\;
    \TantQue{$c \in \mathit{utilisées}$}{
        $c \leftarrow c + 1$ \com{chercher la prochaine couleur libre}
    }
    $\mathit{couleur}[v] \leftarrow c$\;
}
\Retourner $\mathit{couleur}$\;
\end{algorithm}

\subsection{Code Python}

\begin{lstlisting}[style=python,
  caption={Coloration gloutonne — méthode \texttt{\_generate()}}]
# Coloration gloutonne : noeuds de plus fort degré traités en premier
coloring = nx.greedy_color(G, strategy='largest_first')

# Nombre chromatique : plus grande couleur attribuée + 1
self.n_colors = max(coloring.values()) + 1 if coloring else 1
\end{lstlisting}

\begin{tcolorbox}[attention]
\texttt{coloring} est un dictionnaire \texttt{\{nœud: couleur\_entière\}}.
La stratégie \texttt{'largest\_first'} est utilisée pour la génération initiale.
Lors d'une recoloration, d'autres stratégies sont disponibles :
\texttt{'random\_sequential'}, \texttt{'smallest\_last'}, \texttt{'DSATUR'}.
\end{tcolorbox}


\section{Assignation des Rôles aux Nœuds}

\subsection{Explication}

\begin{tcolorbox}[explication]
Après la coloration, chaque couleur entière est \textbf{mappée à un rôle}
du réseau commercial. La couleur est \textbf{plafonnée à 3} : tout nœud
nécessitant une 4\up{e} couleur (ou plus) est automatiquement classé comme
\emph{suspect}, car il brise la structure 3-rôles normale.

\medskip
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|c|c|c}
  \hline
  \textbf{Couleur} & \textbf{Rôle} & \textbf{Forme} & \textbf{Couleur graphe}\\
  \hline
  0 & Consommateur    & \Large●\ cercle   & \textcolor{red!70!black}{\textbf{rouge}} \\
  1 & Fournisseur     & \Large■\ carré    & \textcolor{blue!60!black}{\textbf{bleu}} \\
  2 & Approvisionneur & \Large▲\ triangle & \textcolor{green!60!black}{\textbf{vert}} \\
  $\geq 3$ & Suspect / Fraude & \Large◆\ losange & \textcolor{orange!80!black}{\textbf{orange}} \\
  \hline
\end{tabular}
\end{center}

\medskip
La contrainte de coloration garantit que \textbf{deux nœuds voisins ont
toujours des rôles différents} — ce qui est cohérent avec la séparation
des acteurs dans un réseau commercial réel.
\end{tcolorbox}

\subsection{Algorithme}

\begin{algorithm}[H]
\caption{Assignation des rôles par la coloration}
\KwIn{$\mathit{coloring}$ : dictionnaire $\{$nœud $\to$ couleur$\}$}
\KwOut{$\mathit{roles}$ : dictionnaire $\{$nœud $\to$ rôle $\in \{0,1,2,3\}\}$}
\BlankLine
\Pour{chaque nœud $v$ dans $\mathit{coloring}$}{
    \Si{$\mathit{coloring}[v] \leq 2$}{
        $\mathit{roles}[v] \leftarrow \mathit{coloring}[v]$
        \com{couleur 0, 1 ou 2 = rôle direct}
    }
    \Sinon{
        $\mathit{roles}[v] \leftarrow 3$
        \com{couleur $\geq$ 3 → suspect, hors structure}
    }
}
\Retourner $\mathit{roles}$\;
\end{algorithm}

\subsection{Code Python}

\begin{lstlisting}[style=python,
  caption={Assignation des rôles — méthode \texttt{\_generate()}}]
# Mappage couleur -> role avec plafonnement à 3 pour les suspects
# min(col, 3) : toute couleur >= 3 devient le role "Suspect/Fraude"
self.node_roles = {nd: min(col, 3) for nd, col in coloring.items()}
\end{lstlisting}


\section{Détection de Fraude}

\subsection{Explication}

\begin{tcolorbox}[explication]
Dans un réseau commercial sain, le flux suit obligatoirement la chaîne :
\[
  \underbrace{\textbf{Approvisionneur}}_{\text{source}}
  \;\longrightarrow\;
  \underbrace{\textbf{Fournisseur}}_{\text{intermédiaire}}
  \;\longrightarrow\;
  \underbrace{\textbf{Consommateur}}_{\text{destination}}
\]
Un \textbf{circuit frauduleux} est identifié selon deux critères :

\medskip
\textbf{Critère 1 — Liaison directe interdite :}
Une arête entre un \emph{Consommateur} (rôle 0) et un \emph{Approvisionneur}
(rôle 2) \textbf{sans passer par un Fournisseur} signale un circuit court suspect
(ex. : contournement des intermédiaires légaux, évasion fiscale).

\medskip
\textbf{Critère 2 — Nœud hors structure :}
Tout nœud ayant nécessité une couleur $\geq 3$ lors de la coloration
\textbf{ne s'intègre pas} dans la structure 3-rôles normale.
Il est automatiquement marqué suspect.
\end{tcolorbox}

\subsection{Algorithme}

\begin{algorithm}[H]
\caption{Détection des circuits frauduleux}
\KwIn{$G=(V,E)$, $\mathit{roles}[v]$ : rôle de chaque nœud, $\mathit{coloring}[v]$ : couleur}
\KwOut{$\mathit{fraud\_nodes} \subseteq V$,\quad $\mathit{fraud\_edges} \subseteq E$}
\BlankLine
$\mathit{fraud\_nodes} \leftarrow \varnothing$\;
$\mathit{fraud\_edges} \leftarrow \varnothing$\;
\BlankLine
\com{Critère 1 : arête directe Consommateur (0) \textleftrightarrow\ Approvisionneur (2)}
\Pour{chaque arête $(u,\; v) \in E$}{
    $r_u \leftarrow \mathit{roles}[u]$,\quad $r_v \leftarrow \mathit{roles}[v]$\;
    \Si{$(r_u = 0\; \Et\; r_v = 2)$  \Ou  $(r_u = 2\; \Et\; r_v = 0)$}{
        Ajouter $(u,\; v)$ à $\mathit{fraud\_edges}$\;
        Ajouter $u$ et $v$ à $\mathit{fraud\_nodes}$\;
    }
}
\BlankLine
\com{Critère 2 : nœud nécessitant une couleur $\geq$ 3}
\Pour{chaque nœud $v \in V$}{
    \Si{$\mathit{coloring}[v] > 2$}{
        Ajouter $v$ à $\mathit{fraud\_nodes}$\;
    }
}
\BlankLine
\Retourner $\mathit{fraud\_nodes}$,\; $\mathit{fraud\_edges}$\;
\end{algorithm}

\subsection{Code Python}

\begin{lstlisting}[style=python,
  caption={Détection de fraude — méthode \texttt{\_generate()}}]
# Initialisation des ensembles suspects
self.fraud_nodes     = set()
self.fraud_edges_set = set()

# Critère 1 : liaison directe Consommateur(0) <-> Approvisionneur(2)
for u, v in G.edges():
    ru, rv = self.node_roles[u], self.node_roles[v]
    if (ru == 0 and rv == 2) or (ru == 2 and rv == 0):
        # Stocker l'arête sous forme canonique min(u,v) < max(u,v)
        self.fraud_edges_set.add((min(u, v), max(u, v)))
        self.fraud_nodes.update([u, v])

# Critère 2 : noeud hors structure (couleur >= 3)
for nd, col in coloring.items():
    if col > 2:
        self.fraud_nodes.add(nd)
\end{lstlisting}


\section{Animation Progressive du Graphe}

\subsection{Explication}

\begin{tcolorbox}[explication]
L'animation décompose la construction du graphe en \textbf{4 phases séquentielles},
planifiées avec \texttt{root.after(délai, fonction)} — mécanisme \emph{non bloquant}
de Tkinter qui n'interrompt pas l'interface graphique.

\medskip
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|l|c|c}
  \hline
  \textbf{Phase} & \textbf{Action} & \textbf{Délai} & \textbf{Nb trames}\\
  \hline
  1 & Ajout des nœuds (gris)     & 55 ms & $\lceil n/20 \rceil$ \\
  2 & Ajout des arêtes (grises)  & 35 ms & $\lceil |E|/15 \rceil$ \\
  3 & Coloration des rôles       & 750 ms & 1 \\
  4 & Mise en évidence des fraudes & — & 1 \\
  \hline
\end{tabular}
\end{center}

\medskip
Les nœuds et arêtes sont ajoutés par \textbf{lots} (\textit{batches}) pour éviter
trop de redessins : $b_n = \lfloor n/20 \rfloor$ nœuds par trame,
$b_e = \lfloor |E|/15 \rfloor$ arêtes par trame.
\end{tcolorbox}

\subsection{Algorithme}

\begin{algorithm}[H]
\caption{Animation progressive — Fonction récursive par étapes}
\KwIn{$\mathit{noeuds}$, $\mathit{aretes}$ : listes, $\mathit{etape}$ : entier}
\BlankLine
$b_n \leftarrow \max(1,\; \lfloor|\mathit{noeuds}|/20\rfloor)$
    \com{taille du lot de noeuds}
$b_e \leftarrow \max(1,\; \lfloor|\mathit{aretes}|/15\rfloor)$
    \com{taille du lot d'aretes}
$f_n \leftarrow \lceil|\mathit{noeuds}|/b_n\rceil$
    \com{nombre de trames pour les noeuds}
$f_e \leftarrow \lceil|\mathit{aretes}|/b_e\rceil$
    \com{nombre de trames pour les aretes}
\BlankLine
\Si{$\mathit{etape} < f_n$}{
    \com{Phase 1 : ajout progressif des noeuds (gris)}
    $cnt \leftarrow \min\bigl((\mathit{etape}+1)\times b_n,\;
    |\mathit{noeuds}|\bigr)$\;
    Dessiner noeuds$[0\,{:}\,cnt]$ en gris\;
    Planifier \textsc{Animation}($\mathit{etape}+1$) dans 55 ms\;
}
\SinonSi{$\mathit{etape} < f_n + f_e$}{
    \com{Phase 2 : ajout progressif des aretes (grises)}
    $idx \leftarrow \mathit{etape} - f_n$\;
    $cnt \leftarrow \min\bigl((idx+1)\times b_e,\;
    |\mathit{aretes}|\bigr)$\;
    Dessiner tous les noeuds + aretes$[0\,{:}\,cnt]$ en gris\;
    Planifier \textsc{Animation}($\mathit{etape}+1$) dans 35 ms\;
}
\SinonSi{$\mathit{etape} = f_n + f_e$}{
    \com{Phase 3 : application de la coloration des rôles}
    Dessiner le graphe complet avec les couleurs de rôles\;
    Planifier \textsc{Animation}($\mathit{etape}+1$) dans 750 ms\;
}
\Sinon{
    \com{Phase 4 : mise en évidence des fraudes}
    Dessiner le graphe avec aretes frauduleuses en orange\;
    Ajouter les anneaux pointillés sur les noeuds suspects\;
    Mettre à jour les statistiques et la distribution\;
}
\end{algorithm}

\subsection{Code Python}

\begin{lstlisting}[style=python,
  caption={Animation progressive — méthode \texttt{\_animate()}}]
def _animate(self, nodes: list, edges: list, step: int) -> None:
    n_nd = len(nodes)
    n_ed = len(edges)
    # Taille des lots pour limiter le nombre de redessins
    b_nd = max(1, n_nd // 20)
    b_ed = max(1, n_ed // 15) if n_ed else 1
    # Nombre de trames pour chaque phase
    f_nd = (n_nd + b_nd - 1) // b_nd
    f_ed = (n_ed + b_ed - 1) // b_ed if n_ed else 0

    if step < f_nd:
        # Phase 1 : ajout progressif des noeuds (gris)
        cnt = min((step + 1) * b_nd, n_nd)
        self._draw(nodes[:cnt], [], 'build')
        self.set_status(f"Noeuds : {cnt} / {n_nd}")
        self._after_id = self.root.after(
            55, lambda: self._animate(nodes, edges, step + 1))

    elif step < f_nd + f_ed:
        # Phase 2 : ajout progressif des aretes (grises)
        idx = step - f_nd
        cnt = min((idx + 1) * b_ed, n_ed)
        self._draw(nodes, edges[:cnt], 'build')
        self.set_status(f"Aretes : {cnt} / {n_ed}")
        self._after_id = self.root.after(
            35, lambda: self._animate(nodes, edges, step + 1))

    elif step == f_nd + f_ed:
        # Phase 3 : coloration des roles
        self.set_status("Application de la coloration gloutonne...")
        self._draw(nodes, edges, 'color')
        self._after_id = self.root.after(
            750, lambda: self._animate(nodes, edges, step + 1))

    else:
        # Phase 4 : mise en evidence des fraudes + finalisation
        self.set_status("Detection des fraudes en cours...")
        self._draw(nodes, edges, 'fraud')
        self._update_stats()
        self.busy = False
        self.btn_gen.config(state='normal')
        self.btn_rc.config(state='normal')
\end{lstlisting}
